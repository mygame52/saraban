// ฟังก์ชันการอัปโหลดไฟล์
function uploadImage($file, $target_dir, $filename_base, $index) {
    if (empty($file["name"])) {
        return null; // กรณีที่ไม่มีการอัปโหลดไฟล์
    }

    $imageFileType = strtolower(pathinfo($file["name"], PATHINFO_EXTENSION));
    $target_file = $target_dir . $filename_base . $index . "." . $imageFileType;

    // ตรวจสอบว่าเป็นไฟล์ภาพหรือไม่
    $check = getimagesize($file["tmp_name"]);
    if ($check === false) {
        return null;
    }

    // ตรวจสอบขนาดไฟล์
    if ($file["size"] > 5000000) {  // 5MB
        return null;
    }

    // อนุญาตเฉพาะบางรูปแบบไฟล์
    if ($imageFileType != "jpg" && $imageFileType != "png" && $imageFileType != "jpeg" && $imageFileType != "gif") {
        return null;
    }

    // ตรวจสอบว่าทุกอย่างผ่านหรือไม่
    if (move_uploaded_file($file["tmp_name"], $target_file)) {
        return basename($target_file);
    } else {
        return null;
    }
}
